\subsubsection*{I. Evaluate the integrals or prove their divergence.}

\begin{multicols}{2}
\begin{enumerate}
    \item $\displaystyle \int_{2}^{4} \frac{1}{(x-3)^2}\, dx$
    \item $\displaystyle \int_{0}^{2} \frac{1}{\sqrt{4-x^2}}\, dx$
    \item $\displaystyle \int_{-\infty}^{\infty} \frac{1}{x^2+2x+5}\, dx$
    \item $\displaystyle \int_{e^2}^{\infty} \frac{1}{x\ln^3{x}} \, dx$
    \item $\displaystyle \int_{0}^{\infty} x\sin{x} \, dx$
    \item $\displaystyle \int_{0}^{\infty} e^{-x}\sin{x} \, dx$
\end{enumerate}
\end{multicols}


\subsubsection*{II. Test for convergence using comparison test}
\begin{multicols}{2}
\begin{enumerate}
    \item $\displaystyle \int_{a}^{\infty} e^{-px}\, dx, \quad p>0$
    \item $\displaystyle \int_{0}^{\infty} \frac{1}{1+2x^2+3x^4}\, dx$
    \item $\displaystyle \int_{e^2}^{\infty} \frac{1}{x+\sin^2x} \, dx$
    \item $\displaystyle \int_{1}^{\infty} \frac{\arctan{x}}{x}\, dx$
    \item $\displaystyle \int_{2}^{\infty} \frac{\sqrt[7]{3+2x^2}}{\sqrt[5]{x^3-1}}\, dx$
    \item $\displaystyle \int_{1}^{\infty} 1- \cos{\frac{2}{x}} \, dx$
\end{enumerate}
\end{multicols}
